cmake_minimum_required(VERSION 3.24)
project(urwcn LANGUAGES CXX)
if(NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use the documented bundled i686 MinGW toolchain through build.ps1")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
add_compile_options(-Wall -Wextra -Wpedantic -fno-strict-aliasing)
add_link_options(-static -static-libgcc -static-libstdc++)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
add_executable(uwcn_launcher WIN32 src/launcher/main.cpp)
target_link_options(uwcn_launcher PRIVATE -municode)
target_link_libraries(uwcn_launcher PRIVATE bcrypt shell32)
configure_file(cmake/loader.def "${CMAKE_BINARY_DIR}/loader.def" COPYONLY)
add_library(uwcn_loader SHARED src/loader/host.cpp src/loader/hook_host.cpp src/loader/control_bridge.cpp "${CMAKE_BINARY_DIR}/loader.def")
set_target_properties(uwcn_loader PROPERTIES PREFIX "")
target_link_libraries(uwcn_loader PRIVATE bcrypt)
foreach(module IN ITEMS catalog semantic render layout main_menu adapter)
  file(GLOB_RECURSE module_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/modules/${module}/*.cpp")
  configure_file(cmake/module.def "${CMAKE_BINARY_DIR}/${module}.def" COPYONLY)
  add_library(uwcn_${module} MODULE ${module_sources} "${CMAKE_BINARY_DIR}/${module}.def")
  set_target_properties(uwcn_${module} PROPERTIES PREFIX "")
endforeach()
target_link_libraries(uwcn_render PRIVATE advapi32)
target_link_libraries(uwcn_main_menu PRIVATE bcrypt)
